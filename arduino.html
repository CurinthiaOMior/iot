Conceito

O que é

O Arduino é uma plataforma aberta de prototipagem eletrônica formada por uma placa com microcontrolador e um ambiente de programação (IDE) simples.

É a ponte entre os sensores e o software: a placa lê os pinos de entrada (onde ficam os sensores), processa o programa e comanda as saídas (LEDs, motores, relés). Por ser barato e fácil de programar, é a porta de entrada mais comum para projetos de IoT.

Funcionamento: todo programa (sketch) tem duas partes — setup(), que roda uma vez para configurar os pinos, e loop(), que se repete continuamente lendo sensores e reagindo.

Arduino Uno — características
MicrocontroladorATmega328P
Tensão de operação5 V
Pinos digitais14 (6 com PWM)
Entradas analógicas6 (conversor A/D 10 bits)

Clock16 MHz

Programação

Exemplos lendo sensores

Cada exemplo abaixo é um trecho mínimo que lê um sensor do catálogo e mostra o valor no Monitor Serial. São 10 sensores diferentes.

1 · LDR (luz)

Analógico — leitura direta em A0.

int pino = A0;

void setup() {
  Serial.begin(9600);
}

void loop() {
  int luz = analogRead(pino);   // 0 a 1023
  Serial.println(luz);
  delay(500);
}

2 · DHT22 (temp/umidade)

Biblioteca DHT (Adafruit).

#include <DHT.h>
DHT dht(2, DHT22);

void setup() {
  Serial.begin(9600);
  dht.begin();
}

void loop() {
  float t = dht.readTemperature();
  float u = dht.readHumidity();
  Serial.print(t); Serial.print(" C  ");
  Serial.println(u);
  delay(2000);
}

3 · DS18B20 (temperatura)

Barramento 1-Wire.

#include <OneWire.h>
#include <DallasTemperature.h>
OneWire fio(4);
DallasTemperature s(&fio);

void setup() {
  Serial.begin(9600);
  s.begin();
}

void loop() {
  s.requestTemperatures();
  Serial.println(s.getTempCByIndex(0));
  delay(1000);
}

4 · HC-SR04 (distância)

Mede o tempo do eco.

int trig = 9, echo = 10;

void setup() {
  Serial.begin(9600);
  pinMode(trig, OUTPUT);
  pinMode(echo, INPUT);
}

void loop() {
  digitalWrite(trig, HIGH);
  delayMicroseconds(10);
  digitalWrite(trig, LOW);
  long t = pulseIn(echo, HIGH);
  Serial.println(t / 58);   // cm
  delay(300);
}

5 · PIR (presença)

Saída digital: 1 = movimento.

int pir = 7;

void setup() {
  Serial.begin(9600);
  pinMode(pir, INPUT);
}

void loop() {
  if (digitalRead(pir) == HIGH) {
    Serial.println("Movimento!");
  }
  delay(200);
}

6 · MQ-2 (gás)

Analógico + alarme por limiar.

int pino = A1;

void setup() {
  Serial.begin(9600);
}

void loop() {
  int gas = analogRead(pino);
  Serial.println(gas);
  if (gas > 400) {
    Serial.println("Vazamento!");
  }
  delay(500);
}

7 · BMP280 (pressão)

Comunicação I2C.

#include <Adafruit_BMP280.h>
Adafruit_BMP280 bmp;

void setup() {
  Serial.begin(9600);
  bmp.begin(0x76);
}

void loop() {
  Serial.print(bmp.readPressure() / 100);
  Serial.println(" hPa");
  delay(1000);
}

8 · ACS712 (corrente)

Converte a tensão do sensor em ampères.

int pino = A2;

void setup() {
  Serial.begin(9600);
}

void loop() {
  int leitura = analogRead(pino);
  float volts = leitura * 5.0 / 1023.0;
  float amp = (volts - 2.5) / 0.185;  // 5A
  Serial.println(amp);
  delay(500);
}

9 · MPU-6050 (inercial)

Lê o acelerômetro por I2C.

#include <Wire.h>
#include <MPU6050.h>
MPU6050 mpu;

void setup() {
  Serial.begin(9600);
  Wire.begin();
  mpu.initialize();
}

void loop() {
  int16_t ax, ay, az;
  mpu.getAcceleration(&ax, &ay, &az);
  Serial.println(ax);
  delay(300);
}

10 · Umidade do solo

Analógico: converte para porcentagem.

int pino = A3;

void setup() {
  Serial.begin(9600);
}

void loop() {
  int leitura = analogRead(pino);
  int pct = map(leitura, 1023, 300, 0, 100);
  Serial.print(pct);
  Serial.println(" %");
  delay(1000);
}
